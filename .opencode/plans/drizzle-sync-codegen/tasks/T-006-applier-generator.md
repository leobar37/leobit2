### Generator Design

```typescript
// generators/applier-generator.ts
import type { ColumnMetadata } from "../introspect";
import type { EntityConfig } from "../types";

export interface ApplierConfigOutput {
  validTables: string[];
  tableColumns: Record<string, string[]>;
  requiredDefaults: Record<string, Record<string, unknown>>;
  relationFields: string[];
}

export function generateApplierConfig(
  entityName: string,
  tableName: string,
  columns: ColumnMetadata[],
  config: EntityConfig
): ApplierConfigOutput {
  const columnsToInclude = resolveColumns(columns, config);
  
  // Build column whitelist
  const columnNames = columnsToInclude.map(c => c.name);
  
  // Add sync columns if not present
  if (!columnNames.includes("sync_status")) {
    columnNames.push("sync_status");
  }
  if (!columnNames.includes("sync_attempts")) {
    columnNames.push("sync_attempts");
  }
  
  // Build defaults for required columns
  const defaults: Record<string, unknown> = {};
  
  for (const col of columnsToInclude) {
    if (col.notNull && col.default === undefined) {
      // Column is required but has no default
      // We may need to provide a default during insert
      const defaultValue = inferDefaultForType(col.dataType);
      if (defaultValue !== undefined) {
        defaults[col.name] = defaultValue;
      }
    }
  }
  
  // Always set sync defaults
  defaults["sync_status"] = "synced";
  defaults["sync_attempts"] = 0;
  
  return {
    validTables: [tableName],
    tableColumns: {
      [tableName]: columnNames,
    },
    requiredDefaults: Object.keys(defaults).length > 0 ? {
      [tableName]: defaults,
    } : {},
    relationFields: detectRelationFields(columns),
  };
}

function inferDefaultForType(dataType: string): unknown | undefined {
  switch (dataType) {
    case "string": return "";
    case "number": return 0;
    case "boolean": return false;
    case "date": return new Date().toISOString();
    case "json": return {};
    default: return undefined;
  }
}

function detectRelationFields(columns: ColumnMetadata[]): string[] {
  // Detect fields that are likely relations
  // These would be excluded from direct INSERT/UPDATE
  const relations: string[] = [];
  
  // Common relation field patterns
  const relationPatterns = [
    "items",
    "customer",
    "seller", 
    "business",
    "product",
    "variant",
    "supplier",
  ];
  
  for (const col of columns) {
    // Check if column name matches relation pattern
    if (relationPatterns.some(pattern => 
      camelCase(col.name).toLowerCase() === pattern.toLowerCase()
    )) {
      relations.push(camelCase(col.name));
    }
  }
  
  return relations;
}

// Merge multiple applier configs
export function mergeApplierConfigs(
  configs: ApplierConfigOutput[]
): ApplierConfigOutput {
  const validTables: string[] = [];
  const tableColumns: Record<string, string[]> = {};
  const requiredDefaults: Record<string, Record<string, unknown>> = {};
  const relationFields: string[] = [];
  
  for (const config of configs) {
    validTables.push(...config.validTables);
    Object.assign(tableColumns, config.tableColumns);
    Object.assign(requiredDefaults, config.requiredDefaults);
    relationFields.push(...config.relationFields);
  }
  
  // Deduplicate
  return {
    validTables: [...new Set(validTables)],
    tableColumns,
    requiredDefaults,
    relationFields: [...new Set(relationFields)],
  };
}

// Generate full file content
export function generateApplierFile(config: ApplierConfigOutput): string {
  const lines: string[] = [
    "// ⚠️ AUTO-GENERATED FILE - DO NOT EDIT",
    "// Generated by drizzle-sync from backend schema",
    "",
    "import type { ChangeApplierConfig } from \"@avileo/drizzle-sync/pglite\";",
    "",
    "// Valid table names for change applier",
    `export const VALID_TABLES = new Set([`,
    ...config.validTables.map(t => `  "${t}",`),
    "]);",
    "",
    "// Column whitelist per table",
    "export const TABLE_COLUMNS: Record<string, Set<string>> = {",
  ];
  
  for (const [tableName, columns] of Object.entries(config.tableColumns)) {
    lines.push(`  ${tableName}: new Set([`);
    for (const col of columns) {
      lines.push(`    "${col}",`);
    }
    lines.push("  ]),");
  }
  
  lines.push("};");
  lines.push("");
  
  // Required column defaults
  if (Object.keys(config.requiredDefaults).length > 0) {
    lines.push("// Default values for required columns");
    lines.push("export const REQUIRED_COLUMN_DEFAULTS: Record<string, Record<string, unknown>> = {");
    
    for (const [tableName, defaults] of Object.entries(config.requiredDefaults)) {
      lines.push(`  ${tableName}: {`);
      for (const [col, val] of Object.entries(defaults)) {
        lines.push(`    ${col}: ${JSON.stringify(val)},`);
      }
      lines.push("  },");
    }
    
    lines.push("};");
    lines.push("");
  }
  
  // Relation fields
  if (config.relationFields.length > 0) {
    lines.push("// Fields that represent relations (excluded from INSERT/UPDATE)");
    lines.push(`export const RELATION_FIELDS = new Set([`);
    for (const field of config.relationFields) {
      lines.push(`  "${field}",`);
    }
    lines.push("]);");
    lines.push("");
  }
  
  // Combined config export
  lines.push("// Complete applier config");
  lines.push("export const applierConfig: ChangeApplierConfig = {");
  lines.push("  validTables: VALID_TABLES,");
  lines.push("  tableColumns: TABLE_COLUMNS,");
  if (Object.keys(config.requiredDefaults).length > 0) {
    lines.push("  requiredDefaults: REQUIRED_COLUMN_DEFAULTS,");
  }
  if (config.relationFields.length > 0) {
    lines.push("  relationFields: RELATION_FIELDS,");
  }
  lines.push("};");
  lines.push("");
  lines.push("export default applierConfig;");
  
  return lines.join("\n");
}
```
