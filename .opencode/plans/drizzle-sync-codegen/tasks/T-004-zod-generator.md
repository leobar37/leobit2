### Generator Design

```typescript
// generators/zod-generator.ts
import type { ColumnMetadata } from "../introspect";
import type { EntityConfig } from "../types";

export interface ZodSchemaOutput {
  name: string;              // e.g., "customerSchema"
  schemaCode: string;          // The actual Zod schema code
  typeCode: string;            // The TypeScript type export
}

export function generateZodSchema(
  entityName: string,
  columns: ColumnMetadata[],
  config: EntityConfig
): ZodSchemaOutput {
  const fields: string[] = [];
  
  // Determine which columns to include
  const columnsToInclude = resolveColumns(columns, config);
  
  for (const col of columnsToInclude) {
    const zodCode = columnToZod(col);
    fields.push(`  ${camelCase(col.name)}: ${zodCode},`);
  }
  
  // Add syncStatus if not present
  if (!columnsToInclude.find(c => c.name === "sync_status")) {
    fields.push(`  syncStatus: z.enum(["pending", "synced", "error"]).default("pending"),`);
  }
  
  const schemaCode = `export const ${entityName}Schema = z.object({\n${fields.join("\n")}\n});`;
  
  const typeName = pascalCase(entityName);
  const typeCode = `export type ${typeName} = z.infer<typeof ${entityName}Schema>;`;
  
  return {
    name: `${entityName}Schema`,
    schemaCode,
    typeCode,
  };
}

function columnToZod(col: ColumnMetadata): string {
  let base: string;
  
  switch (col.dataType) {
    case "string":
      base = "z.string()";
      break;
    case "number":
      base = "z.number()";
      break;
    case "boolean":
      base = "z.boolean()";
      break;
    case "date":
      base = "z.coerce.date()";
      break;
    case "enum":
      if (col.enumValues) {
        base = `z.enum([${col.enumValues.map(v => `"${v}"`).join(", ")}])`;
      } else {
        base = "z.string()";
      }
      break;
    case "json":
      base = "z.record(z.unknown())";
      break;
    default:
      base = "z.unknown()";
  }
  
  // Add nullability
  if (!col.notNull) {
    base += ".nullable()";
  }
  
  // Add defaults
  if (col.default !== undefined) {
    base += `.default(${JSON.stringify(col.default)})`;
  }
  
  return base;
}

// Generate full file content
export function generateZodSchemasFile(
  schemas: ZodSchemaOutput[]
): string {
  const imports = `import { z } from "zod";\n\n`;
  
  const content = schemas.map(s => 
    `${s.schemaCode}\n\n${s.typeCode}`
  ).join("\n\n");
  
  const warning = `// ⚠️ AUTO-GENERATED FILE - DO NOT EDIT\n// Generated by drizzle-sync from backend schema\n\n`;
  
  return warning + imports + content;
}
```

### Column Resolution (Hybrid Approach)

```typescript
// Resolve which columns to include based on config
function resolveColumns(
  allColumns: ColumnMetadata[],
  config: EntityConfig
): ColumnMetadata[] {
  // Mode 1: Explicit fields list
  if (config.fields) {
    return allColumns.filter(col => 
      config.fields!.includes(col.name)
    );
  }
  
  // Mode 2: Auto + exclude
  if (config.autoFields && config.excludeFields) {
    return allColumns.filter(col => 
      !config.excludeFields!.includes(col.name)
    );
  }
  
  // Mode 3: Auto all (default)
  return allColumns;
}
```

// ⚠️ AUTO-GENERATED FILE - DO NOT EDIT
// Generated by drizzle-sync from backend schema

import { z } from "zod";

export const customerSchema = z.object({
  id: z.string(),
  name: z.string(),
  dni: z.string().nullable(),
  phone: z.string().nullable(),
  address: z.string().nullable(),
  notes: z.string().nullable(),
  syncStatus: z.enum(["pending", "synced", "error"]).default("pending"),
  syncAttempts: z.number().default(0),
  createdAt: z.coerce.date(),
  updatedAt: z.coerce.date(),
});

export type Customer = z.infer<typeof customerSchema>;

export const saleSchema = z.object({
  id: z.string(),
  businessId: z.string(),
  customerId: z.string().nullable(),
  totalAmount: z.string(),
  status: z.enum(["draft", "confirmed", "active", "delivered", "cancelled"]),
  syncStatus: z.enum(["pending", "synced", "error"]).default("pending"),
  createdAt: z.coerce.date(),
  updatedAt: z.coerce.date(),
});

export type Sale = z.infer<typeof saleSchema>;
