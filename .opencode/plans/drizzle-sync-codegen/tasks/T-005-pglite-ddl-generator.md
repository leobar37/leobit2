### Generator Design

```typescript
// generators/ddl-generator.ts
import type { ColumnMetadata } from "../introspect";
import type { EntityConfig } from "../types";

export interface DDLOutput {
  tableName: string;
  sql: string;
  indexes: string[];
}

export function generateDDL(
  entityName: string,
  tableName: string,
  columns: ColumnMetadata[],
  config: EntityConfig
): DDLOutput {
  const columnsToInclude = resolveColumns(columns, config);
  const columnDefs: string[] = [];
  
  for (const col of columnsToInclude) {
    const def = columnToSQLiteDef(col);
    columnDefs.push(`  ${def}`);
  }
  
  // Add sync columns if not present
  if (!columnsToInclude.find(c => c.name === "sync_status")) {
    columnDefs.push(`  sync_status TEXT NOT NULL DEFAULT 'pending'`);
  }
  if (!columnsToInclude.find(c => c.name === "sync_attempts")) {
    columnDefs.push(`  sync_attempts INTEGER NOT NULL DEFAULT 0`);
  }
  
  // Primary key
  const primaryKeys = columnsToInclude
    .filter(c => c.primary)
    .map(c => c.name);
  
  if (primaryKeys.length > 0) {
    columnDefs.push(`  PRIMARY KEY (${primaryKeys.join(", ")})`);
  }
  
  const sql = `CREATE TABLE IF NOT EXISTS "${tableName}" (\n${columnDefs.join(",\n")}\n);`;
  
  // Generate indexes
  const indexes = generateIndexes(tableName, columnsToInclude);
  
  return {
    tableName,
    sql,
    indexes,
  };
}

function columnToSQLiteDef(col: ColumnMetadata): string {
  const sqliteType = mapToSQLiteType(col.dataType);
  const nullable = col.notNull ? "NOT NULL" : "";
  const defaultVal = col.default !== undefined 
    ? `DEFAULT ${formatDefault(col.default, col.dataType)}` 
    : "";
  
  const parts = [`"${col.name}"`, sqliteType, nullable, defaultVal]
    .filter(Boolean)
    .join(" ");
  
  return parts;
}

function mapToSQLiteType(dataType: string): string {
  switch (dataType) {
    case "string": return "TEXT";
    case "number": return "INTEGER";
    case "boolean": return "INTEGER";  // 0 or 1
    case "date": return "TEXT";       // ISO string
    case "json": return "TEXT";       // JSON string
    case "enum": return "TEXT";
    default: return "TEXT";
  }
}

function formatDefault(value: unknown, dataType: string): string {
  if (value === null) return "NULL";
  if (typeof value === "boolean") return value ? "1" : "0";
  if (typeof value === "number") return String(value);
  if (typeof value === "string") return `'${value.replace(/'/g, "''")}'`;
  return "NULL";
}

function generateIndexes(tableName: string, columns: ColumnMetadata[]): string[] {
  const indexes: string[] = [];
  
  // Always index sync_status for sync queries
  indexes.push(
    `CREATE INDEX IF NOT EXISTS "idx_${tableName}_sync_status" ON "${tableName}"(sync_status);`
  );
  
  // Index business_id for multi-tenancy
  if (columns.find(c => c.name === "business_id")) {
    indexes.push(
      `CREATE INDEX IF NOT EXISTS "idx_${tableName}_business_id" ON "${tableName}"(business_id);`
    );
  }
  
  // Index foreign key columns
  const fkColumns = columns.filter(c => 
    c.name.endsWith("_id") && !c.primary
  );
  
  for (const col of fkColumns) {
    indexes.push(
      `CREATE INDEX IF NOT EXISTS "idx_${tableName}_${col.name}" ON "${tableName}"(${col.name});`
    );
  }
  
  return indexes;
}

// Generate full SQL file
export function generateDDLFile(outputs: DDLOutput[]): string {
  const lines: string[] = [
    "-- ⚠️ AUTO-GENERATED FILE - DO NOT EDIT",
    "-- Generated by drizzle-sync from backend schema",
    "",
    "-- Enable foreign key support",
    "PRAGMA foreign_keys = ON;",
    "",
  ];
  
  for (const output of outputs) {
    lines.push(output.sql);
    lines.push("");
    for (const idx of output.indexes) {
      lines.push(idx);
    }
    lines.push("");
  }
  
  return lines.join("\n");
}
```

## Notes

- SQLite doesn't have native UUID type - use TEXT
- Booleans are INTEGER (0/1) in SQLite
- Dates should be ISO strings (TEXT)
- JSON should be stored as TEXT
- Consider adding foreign key constraints if relations defined
